int add(int a,int b, int c){
    return a+b+c;
}

float power(float base,int exp){
    float res=1.0;
    for(int i=0;i<exp;i++;){
        res*=base;
    }
    return res;
}

bool is_even(int n){
    return (n%2)==0;
}

bool toggle(bool v){
